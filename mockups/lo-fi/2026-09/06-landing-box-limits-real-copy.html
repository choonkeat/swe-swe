Needs swe-swe 2.38.0 or newer.
Save this as your start-up script
#!/bin/sh set -eu : "${SWE_SWE_PASSWORD:?set a password first}" PROJECT_DIR="${PROJECT_DIR:-$HOME/proj}" LOG="$HOME/swe-swe-up.log" curl -fsSL https://raw.githubusercontent.com/choonkeat/swe-swe/main/install.sh | sh export PATH="$HOME/.local/bin:$PATH" mkdir -p "$PROJECT_DIR" cd "$PROJECT_DIR" [ -d .git ] || git init -q swe-swe init --runtime=host --without-mcp export SWE_SWE_PASSWORD SWE_PORT=1977 SWE_AGENT_VIEW=off nohup swe-swe up >"$LOG" 2>&1 & echo "started; log: $LOG"
